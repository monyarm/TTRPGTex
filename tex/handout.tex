\renewcommand{\figurename}{Handout}

\newcounter{handoutCount}
\setcounter{handoutCount}{0}

\makeatletter
\NewDocumentCommand\defineTextHandout{m m m}{
    \defineHandout{#1}{#2}{\begin{tcolorbox}#3\end{tcolorbox}}
}
\NewDocumentCommand\defineCompactTextHandout{m m m}{
    \defineHandout{#1}{#2}{\begingroup\tcbset{before skip=0.1cm,after skip=0.1cm}\begin{tcolorbox}#3\end{tcolorbox}\endgroup}
}
\NewDocumentCommand\defineImageHandout{m m m}{
    \defineHandout{#1}{#2}{\image{#3}}
}
\NewDocumentCommand\defineHandout{m m m}{

    \global\expandafter\gdef\csname handout@#1\endcsname{
        #3
        \noindent\begin{minipage}{\linewidth}
            \captionsetup{labelformat=empty}
            \captionof{figure}{Handout \handoutNum{#1}: #2}
        \end{minipage}
    }
}
\NewDocumentCommand\handout{m}{%

    \ifcsname handout@#1\endcsname%
        \csname handout@#1\endcsname%
    \else%
        \PackageError{handout}{Handout `#1' not defined}{}%
    \fi%
}

\NewDocumentCommand\handoutNum{m}{%
    \ifcsname handoutNum@#1\endcsname%
    \else%
        \stepcounter{handoutCount}%
        \global\expandafter\edef\csname handoutNum@#1\endcsname {\thehandoutCount}
    \fi%
    \ifcsname handoutNum@#1\endcsname%
        \csname handoutNum@#1\endcsname%
    \else%
        \PackageError{handout}{Handout number for `#1' not defined}{}%
    \fi%
}

\NewDocumentCommand\refhandout{m}{%
    (\textbf{Handout \handoutNum{#1}})
}

% \loadHandouts[<folder>]
% Scans <folder> (default: ./handouts) for .tex files and \input each one.
% Also accepts a custom \input prefix when the folder path contains one level
% of nesting that should be used as-is (e.g. \loadHandouts[./new_handouts/handouts]).
\NewDocumentCommand\loadHandouts{O{./handouts}}{%
    \directlua{require("parser").emit_handout_inputs([[#1]])}%
}

\NewDocumentCommand{\allhandouts}{m}{%
\directlua{
    local lfs = require("lfs")
    local excluded = {}
    local raw_excluded = "\luaescapestring{#1}"
    for id in raw_excluded:gmatch("[^, ]+") do excluded[id] = true end

    local files = {}
    for filename in lfs.dir("handouts") do
        if filename:match("[.]tex$") then table.insert(files, filename) end
    end
    table.sort(files)

    local seen, ids = {}, {}
    local slash = string.char(92)
    local percent = string.char(37)
    local open_brace = string.char(123)
    local close_brace = string.char(125)
    local pattern = slash .. "define[" .. percent .. "a]+Handout" .. open_brace .. "([^" .. close_brace .. "]+)" .. close_brace
    for _, filename in ipairs(files) do
        local file = io.open("handouts/" .. filename, "rb")
        if file then
            local contents = file:read("*all")
            file:close()
            for id in contents:gmatch(pattern) do
                if not seen[id] then
                    seen[id] = true
                    table.insert(ids, id)
                end
            end
        end
    end

    local numbers = {}
    for _, id in ipairs(ids) do
        numbers[id] = tonumber(token.get_macro("handoutNum@" .. id)) or math.huge
    end
    table.sort(ids, function(left, right)
        if numbers[left] == numbers[right] then return left < right end
        return numbers[left] < numbers[right]
    end)
    for _, id in ipairs(ids) do
        if not excluded[id] then tex.sprint(slash .. "handout" .. open_brace .. id .. close_brace) end
    end
}
}
\makeatother
